cmake_minimum_required(VERSION 3.20)
project(Evenfall VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(schedule STATIC src/core.cpp)
target_include_directories(schedule PUBLIC src)
target_compile_definitions(schedule PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(platform STATIC src/platform.cpp)
target_link_libraries(platform PUBLIC schedule user32 advapi32 shell32 ole32 uuid bcrypt gdi32)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  target_compile_options(platform PRIVATE /utf-8)
  target_compile_options(schedule PRIVATE /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wno-missing-field-initializers)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()
add_executable(Evenfall WIN32 src/app.cpp resources/app.rc)
target_include_directories(Evenfall PRIVATE resources)
target_link_libraries(Evenfall PRIVATE platform d2d1 dwrite dwmapi comctl32 windowscodecs gdi32)
if(MINGW)
  target_link_options(Evenfall PRIVATE -municode)
endif()
enable_testing()
add_executable(scheduler_tests tests/scheduler_tests.cpp)
target_link_libraries(scheduler_tests PRIVATE schedule)
add_test(NAME scheduler COMMAND scheduler_tests)
add_executable(platform_tests tests/platform_tests.cpp)
target_link_libraries(platform_tests PRIVATE platform)
add_test(NAME persistence_and_platform COMMAND platform_tests test-storage)

if(MINGW)
  target_compile_options(schedule PRIVATE $<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-g1> -ffunction-sections -fdata-sections)
  target_compile_options(platform PRIVATE $<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-g1> -ffunction-sections -fdata-sections)
  target_compile_options(Evenfall PRIVATE $<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-g1> -ffunction-sections -fdata-sections)
  target_link_options(Evenfall PRIVATE -Wl,--gc-sections)
elseif(MSVC)
  target_compile_options(Evenfall PRIVATE $<$<CONFIG:Release>:/Zi>)
  target_link_options(Evenfall PRIVATE $<$<CONFIG:Release>:/DEBUG>)
endif()
